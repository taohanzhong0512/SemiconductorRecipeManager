**实现要点**：
```cpp
// 线程安全的读操作
Recipe RecipeManager::getRecipe(int id) const {
    std::shared_lock<std::shared_mutex> lock(mutex_);
    auto it = recipes_.find(id);
    if (it == recipes_.end()) {
        throw std::invalid_argument("Recipe not found");
    }
    return it->second;
}

// 线程安全的写操作
bool RecipeManager::createRecipe(const Recipe& recipe) {
    std::unique_lock<std::shared_mutex> lock(mutex_);

    // 验证
    QString errorMsg;
    if (!validateRecipe(recipe, errorMsg)) {
        Logger::error("Recipe validation failed: {}", errorMsg.toStdString());
        return false;
    }

    // 分配ID
    Recipe newRecipe = recipe;
    newRecipe.set_id(nextId_++);
    newRecipe.set_created_at(QDateTime::currentSecsSinceEpoch());

    // 保存
    recipes_[newRecipe.id()] = newRecipe;

    // 通知
    emit recipeCreated(newRecipe.id());

    Logger::info("Recipe created: id={}, name={}", newRecipe.id(), newRecipe.name());
    return true;
}

// 配方验证
bool RecipeManager::validateRecipe(const Recipe& recipe, QString& errorMsg) const {
    // 1. 名称不能为空
    if (recipe.name().empty()) {
        errorMsg = "Recipe name cannot be empty";
        return false;
    }

    // 2. 步骤数量检查
    if (recipe.steps_size() == 0) {
        errorMsg = "Recipe must have at least one step";
        return false;
    }

    // 3. 步骤编号连续性检查
    for (int i = 0; i < recipe.steps_size(); ++i) {
        if (recipe.steps(i).step_number() != i + 1) {
            errorMsg = QString("Step number mismatch at index %1").arg(i);
            return false;
        }
    }

    return true;
}
```

// 优化1：预分配缓冲区
class TcpWorker : public QObject {
private:
    QByteArray buffer_;
    static constexpr int kInitialBufferSize = 4096;
    static constexpr int kMaxBufferSize = 1024 * 1024;  // 1MB

public:
    TcpWorker() {
        buffer_.reserve(kInitialBufferSize);
    }
};

// 优化2：错误恢复
void TcpWorker::processBuffer() {
    int errorCount = 0;
    const int kMaxErrors = 3;

    while (buffer_.size() >= sizeof(ProtocolHeader)) {
        ProtocolHeader header;
        memcpy(&header, buffer_.data(), sizeof(header));

        // 魔数校验
        if (header.magic != PROTOCOL_MAGIC) {
            Logger::warn("Invalid magic number, searching for sync...");

            // 尝试重新同步
            bool found = false;
            for (int i = 1; i < buffer_.size() - sizeof(ProtocolHeader); ++i) {
                memcpy(&header, buffer_.data() + i, sizeof(header));
                if (header.magic == PROTOCOL_MAGIC) {
                    buffer_.remove(0, i);
                    found = true;
                    break;
                }
            }

            if (!found) {
                buffer_.clear();
                if (++errorCount >= kMaxErrors) {
                    emit errorOccurred("Too many protocol errors");
                    socket_->disconnectFromHost();
                }
                break;
            }
            continue;
        }

        // 长度校验
        if (header.payloadLen > kMaxBufferSize) {
            Logger::error("Payload too large: {}", header.payloadLen);
            buffer_.clear();
            emit errorOccurred("Payload size exceeded");
            break;
        }

        // ... 正常处理
    }
}
