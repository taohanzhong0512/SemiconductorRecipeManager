```cpp
// tests/common/LoggerTest.cpp
#include <gtest/gtest.h>
#include <filesystem>
#include <fstream>
#include "common/Logger.h"

class LoggerTest : public ::testing::Test {
protected:
    void SetUp() override {
        // 每个测试前清理日志文件
        if (std::filesystem::exists("logs")) {
            std::filesystem::remove_all("logs");
        }
        Logger::init();
    }

    void TearDown() override {
        Logger::shutdown();
    }

    // 辅助函数：读取日志文件内容
    std::string readLogFile(const std::string& filename) {
        std::ifstream file(filename);
        std::stringstream buffer;
        buffer << file.rdbuf();
        return buffer.str();
    }
};

// 测试1：基础日志功能
TEST_F(LoggerTest, BasicLogging) {
    Logger::info("Test info message");
    Logger::warn("Test warn message");
    Logger::error("Test error message");

    // 验证日志文件存在
    EXPECT_TRUE(std::filesystem::exists("logs/app.log"));

    // 验证日志内容
    std::string content = readLogFile("logs/app.log");
    EXPECT_NE(content.find("Test info message"), std::string::npos);
    EXPECT_NE(content.find("Test warn message"), std::string::npos);
    EXPECT_NE(content.find("Test error message"), std::string::npos);
}

// 测试2：日志级别过滤
TEST_F(LoggerTest, LogLevelFilter) {
    Logger::setLogLevel(spdlog::level::warn);

    Logger::debug("Debug message");
    Logger::info("Info message");
    Logger::warn("Warn message");
    Logger::error("Error message");

    std::string content = readLogFile("logs/app.log");

    // debug和info不应该出现
    EXPECT_EQ(content.find("Debug message"), std::string::npos);
    EXPECT_EQ(content.find("Info message"), std::string::npos);

    // warn和error应该出现
    EXPECT_NE(content.find("Warn message"), std::string::npos);
    EXPECT_NE(content.find("Error message"), std::string::npos);
}

// 测试3：多线程日志安全性
TEST_F(LoggerTest, ThreadSafety) {
    const int kThreadCount = 10;
    const int kMessagesPerThread = 100;

    std::vector<std::thread> threads;
    for (int i = 0; i < kThreadCount; ++i) {
        threads.emplace_back([i]() {
            for (int j = 0; j < kMessagesPerThread; ++j) {
                Logger::info("Thread {} message {}", i, j);
            }
        });
    }

    for (auto& t : threads) {
        t.join();
    }

    // 验证所有消息都被记录
    std::string content = readLogFile("logs/app.log");
    int messageCount = 0;
    size_t pos = 0;
    while ((pos = content.find("Thread", pos)) != std::string::npos) {
        ++messageCount;
        ++pos;
    }

    EXPECT_EQ(messageCount, kThreadCount * kMessagesPerThread);
}

// 测试4：日志文件轮转
TEST_F(LoggerTest, LogRotation) {
    // 写入大量日志触发轮转
    for (int i = 0; i < 10000; ++i) {
        Logger::info("Large message with lots of text to trigger rotation {}", i);
    }

    // 验证有多个日志文件
    int logFileCount = 0;
    for (const auto& entry : std::filesystem::directory_iterator("logs")) {
        if (entry.path().extension() == ".log") {
            ++logFileCount;
        }
    }

    EXPECT_GT(logFileCount, 1);
}

// 测试5：性能测试
TEST_F(LoggerTest, Performance) {
    const int kMessageCount = 10000;

    auto start = std::chrono::high_resolution_clock::now();

    for (int i = 0; i < kMessageCount; ++i) {
        Logger::info("Performance test message {}", i);
    }

    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);

    // 10000条日志应该在1秒内完成
    EXPECT_LT(duration.count(), 1000);

    std::cout << "Logged " << kMessageCount << " messages in "
              << duration.count() << "ms" << std::endl;
}
```

```cpp
// tests/core/RecipeManagerTest.cpp
#include <gtest/gtest.h>
#include "core/RecipeManager.h"
#include "proto/recipe.pb.h"

class RecipeManagerTest : public ::testing::Test {
protected:
    void SetUp() override {
        manager_ = &RecipeManager::instance();
        // 清空所有配方
        for (const auto& recipe : manager_->getAllRecipes()) {
            manager_->deleteRecipe(recipe.id());
        }
    }

    RecipeManager* manager_;

    // 辅助函数：创建测试配方
    Recipe createTestRecipe(const std::string& name, int stepCount) {
        Recipe recipe;
        recipe.set_name(name);
        recipe.set_description("Test recipe");
        recipe.set_status(RecipeStatus::DRAFT);

            step->set_step_number(i + 1);

TEST_F(RecipeManagerTest, GetRecipe) {

TEST_F(RecipeManagerTest, UpdateRecipe) {

TEST_F(RecipeManagerTest, DeleteRecipe) {

// 测试5：配方验证
TEST_F(RecipeManagerTest, ValidateRecipe) {
    QString errorMsg;

    // 有效配方
    Recipe validRecipe = createTestRecipe("Valid", 2);
    EXPECT_TRUE(manager_->validateRecipe(validRecipe, errorMsg));

    // 无效配方：空名称
    Recipe invalidRecipe1;
    invalidRecipe1.set_name("");
    EXPECT_FALSE(manager_->validateRecipe(invalidRecipe1, errorMsg));
    EXPECT_NE(errorMsg.toStdString().find("name"), std::string::npos);

    // 无效配方：无步骤
    Recipe invalidRecipe2;
    invalidRecipe2.set_name("No Steps");
    EXPECT_FALSE(manager_->validateRecipe(invalidRecipe2, errorMsg));
    EXPECT_NE(errorMsg.toStdString().find("step"), std::string::npos);

    // 无效配方：步骤编号不连续
    Recipe invalidRecipe3 = createTestRecipe("Bad Steps", 2);
    invalidRecipe3.mutable_steps(1)->set_step_number(5);  // 应该是2
    EXPECT_FALSE(manager_->validateRecipe(invalidRecipe3, errorMsg));
}

// 测试6：查询功能
TEST_F(RecipeManagerTest, FindRecipes) {
    manager_->createRecipe(createTestRecipe("Recipe A", 1));
    manager_->createRecipe(createTestRecipe("Recipe B", 2));
    manager_->createRecipe(createTestRecipe("Test Recipe", 3));

    // 按名称查找
    auto results = manager_->findRecipesByName("Recipe");
    EXPECT_EQ(results.size(), 3);

    results = manager_->findRecipesByName("Test");
    EXPECT_EQ(results.size(), 1);
    EXPECT_EQ(results[0].name(), "Test Recipe");
}

// 测试7：并发测试
TEST_F(RecipeManagerTest, ConcurrentAccess) {
    const int kThreadCount = 10;
    const int kRecipesPerThread = 10;

    // 多线程创建配方
    std::vector<std::thread> threads;
    for (int i = 0; i < kThreadCount; ++i) {
        threads.emplace_back([this, i]() {
            for (int j = 0; j < kRecipesPerThread; ++j) {
                std::string name = "Recipe_" + std::to_string(i) + "_" + std::to_string(j);
                Recipe recipe = createTestRecipe(name, 2);
                manager_->createRecipe(recipe);
            }
        });
    }

    for (auto& t : threads) {
        t.join();
    }

    // 验证所有配方都被创建
    auto recipes = manager_->getAllRecipes();
    EXPECT_EQ(recipes.size(), kThreadCount * kRecipesPerThread);
}

// 测试8：持久化
TEST_F(RecipeManagerTest, Persistence) {
    // 创建配方
    manager_->createRecipe(createTestRecipe("Recipe 1", 2));
    manager_->createRecipe(createTestRecipe("Recipe 2", 3));

    // 保存到文件
    QString filePath = "test_recipes.bin";
    EXPECT_TRUE(manager_->saveToFile(filePath));

    // 清空内存
    for (const auto& recipe : manager_->getAllRecipes()) {
        manager_->deleteRecipe(recipe.id());
    }
    EXPECT_EQ(manager_->getAllRecipes().size(), 0);

    // 从文件加载
    EXPECT_TRUE(manager_->loadFromFile(filePath));
    EXPECT_EQ(manager_->getAllRecipes().size(), 2);

    // 清理
    std::filesystem::remove(filePath.toStdString());
}
```

```cpp
#include <gtest/gtest.h>
#include "gateway/Protocol.h"

TEST(ProtocolTest, HeaderSize) {
    // 验证帧头大小
    EXPECT_EQ(sizeof(ProtocolHeader), 8);
}

TEST(ProtocolTest, MagicNumber) {
    ProtocolHeader header;
    header.magic = PROTOCOL_MAGIC;
    header.payloadLen = 100;

    // 验证魔数
    EXPECT_EQ(header.magic, 0x53454D49);
}

TEST(ProtocolTest, PacketConstruction) {
    QByteArray payload = "Test data";

    // 构造数据包
    QByteArray packet;
    ProtocolHeader header;
    header.magic = PROTOCOL_MAGIC;
    header.payloadLen = payload.size();

    packet.append(reinterpret_cast<const char*>(&header), sizeof(header));
    packet.append(payload);

    // 验证数据包大小
    EXPECT_EQ(packet.size(), sizeof(ProtocolHeader) + payload.size());

    // 解析数据包
    ProtocolHeader parsedHeader;
    memcpy(&parsedHeader, packet.data(), sizeof(parsedHeader));

    EXPECT_EQ(parsedHeader.magic, PROTOCOL_MAGIC);
    EXPECT_EQ(parsedHeader.payloadLen, payload.size());

    QByteArray parsedPayload = packet.mid(sizeof(ProtocolHeader), parsedHeader.payloadLen);
    EXPECT_EQ(parsedPayload, payload);
}

TEST(ProtocolTest, PackAlignment) {
    // 验证没有内存对齐问题
    ProtocolHeader header;
    header.magic = 0x12345678;
    header.payloadLen = 0xABCDEF00;

    char buffer[8];
    memcpy(buffer, &header, sizeof(header));

    ProtocolHeader header2;
    memcpy(&header2, buffer, sizeof(header2));

    EXPECT_EQ(header2.magic, 0x12345678);
    EXPECT_EQ(header2.payloadLen, 0xABCDEF00);
}
```

```cpp
// tests/integration/E2ETest.cpp
#include <gtest/gtest.h>
#include <QTcpSocket>
#include <QEventLoop>
#include "gateway/MesGateway.h"
#include "core/RecipeManager.h"
#include "proto/recipe.pb.h"

class E2ETest : public ::testing::Test {
protected:
    void SetUp() override {
        // 启动服务器
        gateway_ = new MesGateway();
        ASSERT_TRUE(gateway_->startServer(8080));

        // 连接信号
        connect(gateway_, &MesGateway::dataReceived,
                this, &E2ETest::onDataReceived);
    }

    void TearDown() override {
        gateway_->stopServer();
        delete gateway_;
    }

    void onDataReceived(const QByteArray& data) {
        receivedData_ = data;
        dataReceived_ = true;
    }

    MesGateway* gateway_;
    QByteArray receivedData_;
    bool dataReceived_ = false;
};

TEST_F(E2ETest, SendReceiveRecipe) {
    // 创建客户端
    QTcpSocket client;
    client.connectToHost("127.0.0.1", 8080);
    ASSERT_TRUE(client.waitForConnected(1000));

    // 创建配方
    Recipe recipe;
    recipe.set_name("Test Recipe");
    recipe.set_description("E2E Test");

    auto* step = recipe.add_steps();
    step->set_step_number(1);
    step->set_action("Test Action");
    step->set_duration_ms(1000);

    // 序列化
    std::string serialized = recipe.SerializeAsString();

    // 构造协议包
    ProtocolHeader header;
    header.magic = PROTOCOL_MAGIC;
    header.payloadLen = serialized.size();

    QByteArray packet;
    packet.append(reinterpret_cast<const char*>(&header), sizeof(header));
    packet.append(serialized.data(), serialized.size());

    // 发送
    client.write(packet);
    client.flush();

    // 等待接收
    QEventLoop loop;
    QTimer::singleShot(1000, &loop, &QEventLoop::quit);
    loop.exec();

    // 验证
    EXPECT_TRUE(dataReceived_);

    Recipe receivedRecipe;
    EXPECT_TRUE(receivedRecipe.ParseFromArray(receivedData_.data(), receivedData_.size()));
    EXPECT_EQ(receivedRecipe.name(), "Test Recipe");
    EXPECT_EQ(receivedRecipe.steps_size(), 1);
}
```

### 2.2 Python测试客户端

```python
# tests/integration/test_client.py
import socket
import struct
import sys
sys.path.append('proto')
import recipe_pb2

def send_recipe(host, port, recipe):
    """发送配方到服务器"""
    # 序列化
    data = recipe.SerializeToString()

    # 构造协议包
    magic = 0x53454D49
    payload_len = len(data)
    header = struct.pack('<II', magic, payload_len)

    # 发送
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((host, port))
    sock.sendall(header + data)

    # 接收响应
    response = sock.recv(1024)
    sock.close()

    return response

def test_basic_send():
    """测试基础发送功能"""
    recipe = recipe_pb2.Recipe()
    recipe.name = "Python Test Recipe"
    recipe.description = "Created from Python"

    step = recipe.steps.add()
    step.step_number = 1
    step.action = "Test Action"
    step.duration_ms = 1000

    response = send_recipe('localhost', 8080, recipe)
    print(f"Response: {response}")

def test_concurrent_clients():
    """测试并发客户端"""
    import threading

    def client_thread(thread_id):
        recipe = recipe_pb2.Recipe()
        recipe.name = f"Recipe from thread {thread_id}"

        step = recipe.steps.add()
        step.step_number = 1
        step.action = "Concurrent Action"

        send_recipe('localhost', 8080, recipe)
        print(f"Thread {thread_id} completed")

    threads = []
    for i in range(10):
        t = threading.Thread(target=client_thread, args=(i,))
        threads.append(t)
        t.start()

    for t in threads:
        t.join()

    print("All threads completed")

def test_stress():
    """压力测试"""
    import time

    start = time.time()
    count = 100

    for i in range(count):
        recipe = recipe_pb2.Recipe()
        recipe.name = f"Stress Test {i}"

        step = recipe.steps.add()
        step.step_number = 1
        step.action = "Stress Action"

        send_recipe('localhost', 8080, recipe)

    end = time.time()
    duration = end - start

    print(f"Sent {count} recipes in {duration:.2f}s")
    print(f"Throughput: {count/duration:.2f} recipes/s")

if __name__ == '__main__':
    test_basic_send()
    test_concurrent_clients()
    test_stress()
```

## 第五部分：测试最佳实践

### 5.1 测试原则

**FIRST原则**：
- **Fast**：测试应该快速运行
- **Independent**：测试之间相互独立
- **Repeatable**：测试结果可重复
- **Self-Validating**：测试自动验证结果
- **Timely**：及时编写测试

### 5.2 测试覆盖率目标

```
代码覆盖率目标：
├─ 核心业务代码：>80%
├─ 工具类代码：>70%
├─ UI代码：>50%
└─ 整体覆盖率：>70%
```
