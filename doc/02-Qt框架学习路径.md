### 🎯 核心知识点

#### 1.1 Qt对象模型
```cpp
#include <QObject>
#include <QDebug>

// 所有Qt类的基类
class Device : public QObject {
    Q_OBJECT  // 必须！启用元对象系统

    Q_PROPERTY(QString name READ name WRITE setName NOTIFY nameChanged)
    Q_PROPERTY(bool online READ isOnline NOTIFY onlineChanged)

public:
    explicit Device(QObject* parent = nullptr)
        : QObject(parent), online_(false) {}

    QString name() const { return name_; }
    void setName(const QString& name) {
        if (name_ != name) {
            name_ = name;
            emit nameChanged(name_);
        }
    }

    bool isOnline() const { return online_; }

signals:
    void nameChanged(const QString& name);
    void onlineChanged(bool online);

private:
    QString name_;
    bool online_;
};
```

**Qt对象特性**：
- **父子关系**：父对象销毁时自动销毁子对象
- **对象树**：自动内存管理
- **元对象系统**：运行时类型信息（RTTI）
- **属性系统**：Q_PROPERTY宏

#### 1.2 Qt容器类
```cpp
#include <QString>
#include <QVector>
#include <QMap>
#include <QList>

// QString：强大的字符串类
QString str = "Hello";
str.append(" World");
str.toUpper();  // "HELLO WORLD"
QStringList parts = str.split(" ");  // ["HELLO", "WORLD"]

// QVector：类似std::vector
QVector<int> numbers = {1, 2, 3, 4, 5};
numbers.append(6);

}

    qDebug() << name;
}
```

**Qt容器 vs STL容器**：
- Qt容器：隐式共享（写时复制），Qt风格API
- STL容器：标准C++，更广泛支持
- **建议**：Qt API用Qt容器，内部逻辑可用STL

#### 1.3 Qt文件I/O
```cpp
#include <QFile>
#include <QTextStream>
#include <QDataStream>
#include <QDir>

// 文本文件读写
void readTextFile(const QString& path) {
    QFile file(path);
    if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
        qWarning() << "Cannot open file:" << path;
        return;
    }

    QTextStream in(&file);
    while (!in.atEnd()) {
        QString line = in.readLine();
        qDebug() << line;
    }
    file.close();
}

void writeTextFile(const QString& path, const QString& content) {
    QFile file(path);
    if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
        qWarning() << "Cannot write file:" << path;
        return;
    }

    QTextStream out(&file);
    out << content;
    file.close();
}

// 二进制文件读写
void writeBinary(const QString& path, const QByteArray& data) {
    QFile file(path);
    if (file.open(QIODevice::WriteOnly)) {
        QDataStream out(&file);
        out << data;
    }
}

// 目录操作
void listFiles(const QString& dirPath) {
    QDir dir(dirPath);
    QStringList files = dir.entryList(QDir::Files);
    for (const QString& file : files) {
        qDebug() << file;
    }
}
```

```cpp
// 1. 新式连接（推荐）：编译时检查
```

#### 3.2 线程安全
```cpp
#include <QMutex>
#include <QMutexLocker>
#include <QReadWriteLock>

// 互斥锁
class ThreadSafeQueue {
private:
    QQueue<int> queue_;
    mutable QMutex mutex_;

public:
    void enqueue(int value) {
        QMutexLocker locker(&mutex_);  // RAII锁
        queue_.enqueue(value);
    }

    int dequeue() {
        QMutexLocker locker(&mutex_);
        return queue_.dequeue();
    }

    int size() const {
        QMutexLocker locker(&mutex_);
        return queue_.size();
    }
};

// 读写锁（多读单写）
class Cache {
private:
    QMap<QString, QString> data_;
    mutable QReadWriteLock lock_;

public:
    QString get(const QString& key) const {
        QReadLocker locker(&lock_);  // 读锁
        return data_.value(key);
    }

    void set(const QString& key, const QString& value) {
        QWriteLocker locker(&lock_);  // 写锁
        data_[key] = value;
    }
};
```

```cpp
class TcpWorker : public QObject {
    Q_OBJECT

private:
    QTcpSocket* socket_;
    QByteArray buffer_;

public:
    TcpWorker(qintptr socketDescriptor) {
        socket_ = new QTcpSocket(this);
        socket_->setSocketDescriptor(socketDescriptor);

        // 信号槽在同一线程，DirectConnection
        connect(socket_, &QTcpSocket::readyRead,
                this, &TcpWorker::onReadyRead);

private slots:
    void onReadyRead() {
        // 在子线程中执行
        buffer_.append(socket_->readAll());
        processBuffer();
    }

    void processBuffer() {
        // 粘包处理逻辑
        while (buffer_.size() >= 8) {
            // ... 解析协议
            emit dataReceived(data);  // 跨线程发送到主线程
        }
    }

signals:
    void dataReceived(const QByteArray& data);
};
```

#### 4.3 粘包处理（项目核心）
```cpp
// src/gateway/Protocol.h
#pragma pack(push, 1)
struct ProtocolHeader {
    uint32_t magic;      // 0x53454D49
    uint32_t payloadLen;
};
#pragma pack(pop)

// src/gateway/TcpWorker.cpp
void TcpWorker::processBuffer() {
    while (true) {
        // 1. 检查是否有完整帧头
        if (buffer_.size() < sizeof(ProtocolHeader)) {
            break;
        }

        // 2. 解析帧头
        ProtocolHeader header;
        memcpy(&header, buffer_.data(), sizeof(header));

        // 3. 校验魔数
        if (header.magic != 0x53454D49) {
            qWarning() << "Invalid magic number";
            buffer_.clear();
            break;
        }

        // 4. 检查是否有完整数据
        uint32_t totalLen = sizeof(ProtocolHeader) + header.payloadLen;
        if (buffer_.size() < totalLen) {
            break;  // 等待更多数据
        }

        // 5. 提取完整帧
        QByteArray frame = buffer_.mid(sizeof(ProtocolHeader), header.payloadLen);
        emit dataReceived(frame);

        // 6. 移除已处理数据
        buffer_.remove(0, totalLen);
    }
}
```

```cpp
import socket
import struct

def send_recipe(host, port, recipe_data):
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((host, port))

    # 构造协议帧
    magic = 0x53454D49
    payload_len = len(recipe_data)
```

#### 5.3 Model/View模式
```cpp
// C++: 自定义模型
class RecipeModel : public QAbstractListModel {
    Q_OBJECT

public:
    enum Roles {
        NameRole = Qt::UserRole + 1,
        StepCountRole
    };

    int rowCount(const QModelIndex& parent = QModelIndex()) const override {
        return recipes_.size();
    }

    QVariant data(const QModelIndex& index, int role) const override {
        if (!index.isValid()) return QVariant();

        const Recipe& recipe = recipes_[index.row()];
        switch (role) {
            case NameRole: return recipe.name();
            case StepCountRole: return recipe.steps_size();
        }
        return QVariant();
    }

    QHash<int, QByteArray> roleNames() const override {
        return {
            {NameRole, "name"},
            {StepCountRole, "stepCount"}
        };
    }

private:
    std::vector<Recipe> recipes_;
};

// QML: 使用模型
ListView {
    model: recipeModel
    delegate: Text {
        text: model.name + " (" + model.stepCount + " steps)"
```
