#### 基础SQL操作

```sql
-- 创建数据库
CREATE DATABASE semiconductor_db
    WITH ENCODING='UTF8'
    LC_COLLATE='C'
    LC_CTYPE='C'
    TEMPLATE=template0;

-- 连接数据库
\c semiconductor_db

-- 创建配方表
CREATE TABLE recipes (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    status VARCHAR(50) DEFAULT 'DRAFT',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    data BYTEA  -- 存储Protobuf序列化数据
);

-- 创建配方步骤表
CREATE TABLE recipe_steps (
    id SERIAL PRIMARY KEY,
    recipe_id INTEGER REFERENCES recipes(id) ON DELETE CASCADE,
    step_number INTEGER NOT NULL,
    action VARCHAR(255) NOT NULL,
    duration_ms INTEGER,
    parameters JSONB,  -- 使用JSONB存储参数
    description TEXT,
    UNIQUE(recipe_id, step_number)
);

-- 创建索引
CREATE INDEX idx_recipes_name ON recipes(name);
CREATE INDEX idx_recipes_status ON recipes(status);
CREATE INDEX idx_recipe_steps_recipe_id ON recipe_steps(recipe_id);

-- 创建设备表
CREATE TABLE devices (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    type VARCHAR(100) NOT NULL,
    ip_address INET,
    port INTEGER,
    status VARCHAR(50) DEFAULT 'OFFLINE',
    last_seen TIMESTAMP,
    metadata JSONB
);

-- 创建设备状态历史表
CREATE TABLE device_status_history (
    id SERIAL PRIMARY KEY,
    device_id INTEGER REFERENCES devices(id) ON DELETE CASCADE,
    status VARCHAR(50) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    details JSONB
);

-- 创建用户表
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(50) DEFAULT 'operator',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMP
);

-- 创建操作日志表
CREATE TABLE operation_logs (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id),
    operation VARCHAR(100) NOT NULL,
    target_type VARCHAR(50),
    target_id INTEGER,
    details JSONB,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 创建触发器：自动更新updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_recipes_updated_at
    BEFORE UPDATE ON recipes
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();
```

```cpp
// src/database/DatabaseManager.cpp
#include "database/DatabaseManager.h"
#include "common/Logger.h"
#include "common/ConfigManager.h"
#include <fstream>
#include <sstream>

DatabaseManager& DatabaseManager::instance() {
    static DatabaseManager instance;
    return instance;
}

bool DatabaseManager::initialize(const QString& configPath) {
    std::lock_guard<std::mutex> lock(mutex_);

    try {
        // 读取配置
        auto& config = ConfigManager::instance();
        QString host = config.getValue("database.host", "localhost").toString();
        int port = config.getValue("database.port", 5432).toInt();
        QString dbname = config.getValue("database.dbname", "semiconductor_db").toString();
        QString user = config.getValue("database.user", "postgres").toString();
        QString password = config.getValue("database.password", "").toString();

        // 构造连接字符串
        std::ostringstream oss;
        oss << "host=" << host.toStdString()
            << " port=" << port
            << " dbname=" << dbname.toStdString()
            << " user=" << user.toStdString()
            << " password=" << password.toStdString()
            << " connect_timeout=10";

        connectionString_ = oss.str();

        // 测试连接
        pqxx::connection conn(connectionString_);
        if (!conn.is_open()) {
            Logger::error("Failed to connect to database");
            return false;
        }

        Logger::info("Database connection established: {}", dbname.toStdString());
        initialized_ = true;
        return true;

    } catch (const std::exception& e) {
        Logger::error("Database initialization failed: {}", e.what());
        return false;
    }
}

std::shared_ptr<pqxx::connection> DatabaseManager::getConnection() {
    if (!initialized_) {
        throw std::runtime_error("DatabaseManager not initialized");
    }

    try {
        auto conn = std::make_shared<pqxx::connection>(connectionString_);
        if (!conn->is_open()) {
            throw std::runtime_error("Failed to open database connection");
        }
        return conn;

    } catch (const std::exception& e) {
        Logger::error("Failed to get database connection: {}", e.what());
        throw;
    }
}

bool DatabaseManager::executeScript(const QString& scriptPath) {
    try {
        // 读取SQL脚本
        std::ifstream file(scriptPath.toStdString());
        if (!file.is_open()) {
            Logger::error("Failed to open SQL script: {}", scriptPath.toStdString());
            return false;
        }

        std::stringstream buffer;
        buffer << file.rdbuf();
        std::string sql = buffer.str();

        // 执行脚本
        auto conn = getConnection();
        pqxx::work txn(*conn);
        txn.exec(sql);
        txn.commit();

        Logger::info("SQL script executed successfully: {}", scriptPath.toStdString());
        return true;

    } catch (const std::exception& e) {
        Logger::error("Failed to execute SQL script: {}", e.what());
        return false;
    }
}

bool DatabaseManager::isHealthy() const {
    try {
        auto conn = const_cast<DatabaseManager*>(this)->getConnection();
        pqxx::nontransaction txn(*conn);
        txn.exec("SELECT 1");
        return true;

    } catch (const std::exception& e) {
        Logger::warn("Database health check failed: {}", e.what());
        return false;
    }
}

void DatabaseManager::shutdown() {
    std::lock_guard<std::mutex> lock(mutex_);
    initialized_ = false;
    Logger::info("Database manager shutdown");
}

DatabaseManager::~DatabaseManager() {
    shutdown();
}
```

### 2.5 连接池实现

```cpp
// include/database/ConnectionPool.h
#pragma once

#include <pqxx/pqxx>
#include <memory>
#include <queue>
#include <mutex>
#include <condition_variable>
#include <chrono>

class ConnectionPool {
public:
    static ConnectionPool& instance();

    // 初始化连接池
    bool initialize(int minConnections, int maxConnections);

    // 获取连接（RAII包装）
    class ConnectionGuard {
    public:
        ConnectionGuard(std::shared_ptr<pqxx::connection> conn, ConnectionPool* pool);
        ~ConnectionGuard();

        pqxx::connection* operator->() { return conn_.get(); }
        pqxx::connection& operator*() { return *conn_; }

    private:
        std::shared_ptr<pqxx::connection> conn_;
        ConnectionPool* pool_;
    };

    ConnectionGuard getConnection(std::chrono::milliseconds timeout = std::chrono::seconds(30));

    // 统计信息
    int activeConnections() const;
    int idleConnections() const;

private:
    ConnectionPool() = default;
    ~ConnectionPool();

    void returnConnection(std::shared_ptr<pqxx::connection> conn);
    std::shared_ptr<pqxx::connection> createConnection();

    std::queue<std::shared_ptr<pqxx::connection>> pool_;
    mutable std::mutex mutex_;
    std::condition_variable cv_;

    int minConnections_ = 5;
    int maxConnections_ = 20;
    int activeCount_ = 0;
};
```

```cpp
// src/database/ConnectionPool.cpp
#include "database/ConnectionPool.h"
#include "database/DatabaseManager.h"
#include "common/Logger.h"

ConnectionPool& ConnectionPool::instance() {
    static ConnectionPool instance;
    return instance;
}

bool ConnectionPool::initialize(int minConnections, int maxConnections) {
    std::lock_guard<std::mutex> lock(mutex_);

    minConnections_ = minConnections;
    maxConnections_ = maxConnections;

    // 创建最小连接数
    for (int i = 0; i < minConnections_; ++i) {
        try {
            auto conn = createConnection();
            pool_.push(conn);
        } catch (const std::exception& e) {
            Logger::error("Failed to create initial connection: {}", e.what());
            return false;
        }
    }

    Logger::info("Connection pool initialized: min={}, max={}", minConnections_, maxConnections_);
    return true;
}

std::shared_ptr<pqxx::connection> ConnectionPool::createConnection() {
    return DatabaseManager::instance().getConnection();
}

ConnectionPool::ConnectionGuard ConnectionPool::getConnection(std::chrono::milliseconds timeout) {
    std::unique_lock<std::mutex> lock(mutex_);

    // 等待可用连接
    if (!cv_.wait_for(lock, timeout, [this]() {
        return !pool_.empty() || activeCount_ < maxConnections_;
    })) {
        throw std::runtime_error("Connection pool timeout");
    }

    std::shared_ptr<pqxx::connection> conn;

    if (!pool_.empty()) {
        // 从池中获取
        conn = pool_.front();
        pool_.pop();
    } else {
        // 创建新连接
        conn = createConnection();
    }

    ++activeCount_;
    return ConnectionGuard(conn, this);
}

void ConnectionPool::returnConnection(std::shared_ptr<pqxx::connection> conn) {
    std::lock_guard<std::mutex> lock(mutex_);

    --activeCount_;

    // 检查连接是否健康
    if (conn && conn->is_open()) {
        pool_.push(conn);
    } else {
        Logger::warn("Discarding unhealthy connection");
    }

    cv_.notify_one();
}

int ConnectionPool::activeConnections() const {
    std::lock_guard<std::mutex> lock(mutex_);
    return activeCount_;
}

int ConnectionPool::idleConnections() const {
    std::lock_guard<std::mutex> lock(mutex_);
    return pool_.size();
}

ConnectionPool::~ConnectionPool() {
    std::lock_guard<std::mutex> lock(mutex_);
    while (!pool_.empty()) {
        pool_.pop();
    }
    Logger::info("Connection pool destroyed");
}

// ConnectionGuard实现
ConnectionPool::ConnectionGuard::ConnectionGuard(
    std::shared_ptr<pqxx::connection> conn, ConnectionPool* pool)
    : conn_(conn), pool_(pool) {}

ConnectionPool::ConnectionGuard::~ConnectionGuard() {
    if (conn_ && pool_) {
        pool_->returnConnection(conn_);
    }
}
```

```cpp
// src/database/RecipeDao.cpp
#include "database/RecipeDao.h"
#include "database/ConnectionPool.h"
#include "common/Logger.h"

int RecipeDao::create(const Recipe& recipe) {
    try {
        auto conn = ConnectionPool::instance().getConnection();
        pqxx::work txn(**conn);

        // 序列化Protobuf数据
        std::string serialized = recipe.SerializeAsString();

        // 插入配方
        std::string sql = R"(
            INSERT INTO recipes (name, description, status, data)
            VALUES ($1, $2, $3, $4)
            RETURNING id
        )";

        pqxx::result result = txn.exec_params(
            sql,
            recipe.name(),
            recipe.description(),
            static_cast<int>(recipe.status()),
            pqxx::binarystring(serialized.data(), serialized.size())
        );

        int recipeId = result[0][0].as<int>();

        // 插入配方步骤
        for (int i = 0; i < recipe.steps_size(); ++i) {
            const auto& step = recipe.steps(i);

            std::string stepSql = R"(
                INSERT INTO recipe_steps (recipe_id, step_number, action, duration_ms, description)
                VALUES ($1, $2, $3, $4, $5)
            )";

            txn.exec_params(
                stepSql,
                recipeId,
                step.step_number(),
                step.action(),
                step.duration_ms(),
                step.description()
            );
        }

        txn.commit();

        Logger::info("Recipe created: id={}, name={}", recipeId, recipe.name());
        return recipeId;

    } catch (const std::exception& e) {
        Logger::error("Failed to create recipe: {}", e.what());
        throw;
    }
}

std::optional<Recipe> RecipeDao::findById(int id) {
    try {
        auto conn = ConnectionPool::instance().getConnection();
        pqxx::nontransaction txn(**conn);

        std::string sql = "SELECT * FROM recipes WHERE id = $1";
        pqxx::result result = txn.exec_params(sql, id);

        if (result.empty()) {
            return std::nullopt;
        }

        return parseRecipe(result[0]);

    } catch (const std::exception& e) {
        Logger::error("Failed to find recipe by id: {}", e.what());
        throw;
    }
}

QVector<Recipe> RecipeDao::findAll() {
    try {
        auto conn = ConnectionPool::instance().getConnection();
        pqxx::nontransaction txn(**conn);

        std::string sql = "SELECT * FROM recipes ORDER BY created_at DESC";
        pqxx::result result = txn.exec(sql);

        QVector<Recipe> recipes;
        for (const auto& row : result) {
            recipes.push_back(parseRecipe(row));
        }

        return recipes;

    } catch (const std::exception& e) {
        Logger::error("Failed to find all recipes: {}", e.what());
        throw;
    }
}

QVector<Recipe> RecipeDao::findByName(const QString& keyword) {
    try {
        auto conn = ConnectionPool::instance().getConnection();
        pqxx::nontransaction txn(**conn);

        std::string sql = "SELECT * FROM recipes WHERE name ILIKE $1 ORDER BY created_at DESC";
        std::string pattern = "%" + keyword.toStdString() + "%";
        pqxx::result result = txn.exec_params(sql, pattern);

        QVector<Recipe> recipes;
        for (const auto& row : result) {
            recipes.push_back(parseRecipe(row));
        }

        return recipes;

    } catch (const std::exception& e) {
        Logger::error("Failed to find recipes by name: {}", e.what());
        throw;
    }
}

bool RecipeDao::update(const Recipe& recipe) {
    try {
        auto conn = ConnectionPool::instance().getConnection();
        pqxx::work txn(**conn);

        // 序列化Protobuf数据
        std::string serialized = recipe.SerializeAsString();

        std::string sql = R"(
            UPDATE recipes
            SET name = $1, description = $2, status = $3, data = $4
            WHERE id = $5
        )";

        pqxx::result result = txn.exec_params(
            sql,
            recipe.name(),
            recipe.description(),
            static_cast<int>(recipe.status()),
            pqxx::binarystring(serialized.data(), serialized.size()),
            recipe.id()
        );

        txn.commit();

        Logger::info("Recipe updated: id={}", recipe.id());
        return result.affected_rows() > 0;

    } catch (const std::exception& e) {
        Logger::error("Failed to update recipe: {}", e.what());
        throw;
    }
}

bool RecipeDao::deleteById(int id) {
    try {
        auto conn = ConnectionPool::instance().getConnection();
        pqxx::work txn(**conn);

        std::string sql = "DELETE FROM recipes WHERE id = $1";
        pqxx::result result = txn.exec_params(sql, id);

        txn.commit();

        Logger::info("Recipe deleted: id={}", id);
        return result.affected_rows() > 0;

    } catch (const std::exception& e) {
        Logger::error("Failed to delete recipe: {}", e.what());
        throw;
    }
}

Recipe RecipeDao::parseRecipe(const pqxx::row& row) {
    Recipe recipe;

    // 从BYTEA字段反序列化
    pqxx::binarystring data(row["data"]);
    recipe.ParseFromArray(data.data(), data.size());

    // 设置其他字段
    recipe.set_id(row["id"].as<int>());
    recipe.set_name(row["name"].as<std::string>());
    recipe.set_description(row["description"].as<std::string>());
    recipe.set_status(static_cast<RecipeStatus>(row["status"].as<int>()));

    return recipe;
}

int RecipeDao::count() {
    try {
        auto conn = ConnectionPool::instance().getConnection();
        pqxx::nontransaction txn(**conn);

        pqxx::result result = txn.exec("SELECT COUNT(*) FROM recipes");
        return result[0][0].as<int>();

    } catch (const std::exception& e) {
        Logger::error("Failed to count recipes: {}", e.what());
        throw;
    }
}
```

```cpp
#include <gtest/gtest.h>
#include "database/RecipeDao.h"
#include "database/DatabaseManager.h"

class RecipeDaoTest : public ::testing::Test {
protected:
    void SetUp() override {
        // 使用测试数据库
        DatabaseManager::instance().initialize("config/test_database.toml");
        ConnectionPool::instance().initialize(2, 5);

        // 清空表
        auto conn = ConnectionPool::instance().getConnection();
        pqxx::work txn(**conn);
        txn.exec("TRUNCATE recipes CASCADE");
        txn.commit();
    }
};

TEST_F(RecipeDaoTest, CreateAndFind) {
    RecipeDao dao;

    Recipe recipe;
    recipe.set_name("Test Recipe");
    recipe.set_description("Test");

    int id = dao.create(recipe);
    EXPECT_GT(id, 0);

    auto found = dao.findById(id);
    ASSERT_TRUE(found.has_value());
    EXPECT_EQ(found->name(), "Test Recipe");
}
```
